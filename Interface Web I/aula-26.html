Web Storage: A Memória do Navegador

Toda lista de tarefas que fizemos até agora sumia ao recarregar a página. Hoje isso acaba — vamos guardar dados direto no navegador do usuário, sem precisar de servidor nenhum.

1 Conceitos-Chave

O navegador tem um "cofre" próprio pra cada site, onde dá pra guardar informação que sobrevive a um recarregamento.

  • localStorage guarda dados que sobrevivem até serem apagados — mesmo fechando o navegador.
  • sessionStorage funciona igual, mas os dados somem ao fechar a ABA (sobrevive a um F5, não a uma aba nova).
  • localStorage.setItem('chave', valor) salva; .getItem('chave') lê de volta (ou null se não existir).
  • .removeItem('chave') apaga uma chave; .clear() apaga TUDO daquele site.

Só Aceita Texto!

setItem só guarda STRINGS. Pra salvar um objeto ou array, use JSON.stringify() (Aula 25) antes de salvar, e JSON.parse() depois de ler — senão você recebe de volta o texto "[object Object]", inútil.

Salvando um Objeto na Prática

Para salvar

localStorage.setItem('tarefas', JSON.stringify(array));

Para ler de volta

const array = JSON.parse(localStorage.getItem('tarefas'));

2. Exemplo Prático: Lista de Tarefas Persistente

A mesma lista de tarefas de sempre — mas agora, se você recarregar a página (F5), ela continua lá.

lista_persistente.html 
<style>
  .painel { display: grid; grid-template-columns: 1fr auto; gap: 8px; background: #fff; padding: 20px; border-radius: 12px; width: 300px; }
  #lista { grid-column: 1 / 3; list-style: none; padding: 0; margin: 10px 0 0; }
  #lista li { display: flex; justify-content: space-between; padding: 6px 0; border-bottom: 1px solid #f1f5f9; }
</style>

<div class="painel">
  <input id="input-tarefa" placeholder="Nova tarefa...">
  <button id="btn-add">+</button>
  <ul id="lista"></ul>
</div>

<script>
  const inputTarefa = document.querySelector('#input-tarefa');
  const botao = document.querySelector('#btn-add');
  const lista = document.querySelector('#lista');

  function carregarTarefas() {
    const dados = localStorage.getItem('tarefas');
    return dados ? JSON.parse(dados) : [];   // primeira vez: array vazio
  }

  function salvarTarefas() {
    localStorage.setItem('tarefas', JSON.stringify(tarefas));
  }

  let tarefas = carregarTarefas();          // já carrega o que tinha salvo

  function renderizar() {
    lista.innerHTML = '';
    tarefas.forEach((tarefa, indice) => {
      const li = document.createElement('li');
      li.textContent = tarefa;

      const botaoRemover = document.createElement('button');
      botaoRemover.textContent = '✕';
      botaoRemover.addEventListener('click', () => {
        tarefas.splice(indice, 1);
        salvarTarefas();
        renderizar();
      });

      li.appendChild(botaoRemover);
      lista.appendChild(li);
    });
  }

  botao.addEventListener('click', () => {
    if (inputTarefa.value) {
      tarefas.push(inputTarefa.value);
      salvarTarefas();
      inputTarefa.value = '';
      renderizar();
    }
  });

  renderizar();
  </script>
Resultado Renderizado

Adicione uma tarefa e dê F5 no navegador de verdade (fora deste preview) pra confirmar que ela continua lá.

Exercício 1 — Salvar Preferência Simples

⭐

O que você tem

<select id="select-tema">
  <option value="claro">Claro</option>
  <option value="escuro">Escuro</option>
</select>
<button id="btn-salvar">Salvar</button>
<button id="btn-carregar">Carregar Salva</button>
<p id="saida">--</p>

<script>
  const selectTema = document.querySelector('#select-tema');
  const saida = document.querySelector('#saida');
  // seu código aqui
</script>

O que fazer

"Salvar" guarda selectTema.value em localStorage. "Carregar Salva" lê de volta e mostra.

Dica

Como o valor já é uma string, não precisa de JSON.stringify aqui.

Solução

document.querySelector('#btn-salvar').addEventListener('click', () => {
  localStorage.setItem('tema', selectTema.value);
  saida.textContent = 'Salvo: ' + selectTema.value;
});

document.querySelector('#btn-carregar').addEventListener('click', () => {
  const tema = localStorage.getItem('tema');
  saida.textContent = tema ? 'Tema salvo: ' + tema : 'Nada salvo ainda.';
});

Resultado esperado

Prévia

Tema salvo: escuro

Exercício 2 — Guardar um Objeto

⭐⭐

O que você tem

<input id="nome" value="Neo">
<input id="idade" value="25">
<button id="btn-salvar">Salvar Perfil</button>
<button id="btn-ler">Ler Perfil Salvo</button>
<p id="saida">--</p>

<script>
  const inputNome = document.querySelector('#nome');
  const inputIdade = document.querySelector('#idade');
  const saida = document.querySelector('#saida');
  // seu código aqui
</script>

O que fazer

"Salvar" cria um objeto { nome, idade } e guarda com JSON.stringify. "Ler" busca, converte com JSON.parse, e mostra.

Dica

Sem JSON.stringify, o objeto vira o texto inútil "[object Object]".

Solução

document.querySelector('#btn-salvar').addEventListener('click', () => {
  const perfil = { nome: inputNome.value, idade: Number(inputIdade.value) };
  localStorage.setItem('perfil', JSON.stringify(perfil));
  saida.textContent = 'Perfil salvo!';
});

document.querySelector('#btn-ler').addEventListener('click', () => {
  const dados = localStorage.getItem('perfil');
  if (!dados) { saida.textContent = 'Nenhum perfil salvo.'; return; }
  const perfil = JSON.parse(dados);
  saida.textContent = perfil.nome + ', ' + perfil.idade + ' anos';
});

Resultado esperado

Prévia

Neo, 25 anos

<input id="input-tarefa" placeholder="Nova tarefa...">

<script>

O que fazer

Solução

function carregarTarefas() {
  const dados = localStorage.getItem('tarefas-ex3');

}

let tarefas = carregarTarefas();

function renderizar() {
  lista.innerHTML = '';
  tarefas.forEach((tarefa) => {
    const li = document.createElement('li');
    li.textContent = tarefa;
    lista.appendChild(li);
  });
}

botao.addEventListener('click', () => {
  if (inputTarefa.value) {
    tarefas.push(inputTarefa.value);
    localStorage.setItem('tarefas-ex3', JSON.stringify(tarefas));
    inputTarefa.value = '';
    renderizar();
  }
});

renderizar();

Exercício 5 — Migrando o To-Do da Aula 23

⭐⭐⭐⭐⭐

O que você tem

<form id="form-tarefa">
  <input id="input-tarefa" placeholder="Nova tarefa...">
  <button type="submit">Adicionar</button>
</form>
<ul id="lista-tarefas"></ul>

<script>
  const form = document.querySelector('#form-tarefa');
  const inputTarefa = document.querySelector('#input-tarefa');
  const listaTarefas = document.querySelector('#lista-tarefas');
  // seu código aqui
</script>

O que fazer

1. Guarde cada tarefa como { texto, concluida: false } (não só o texto) — pra lembrar quais estão feitas.

2. Carregue do localStorage no início; salve toda vez que mudar.

3. Use submit + preventDefault (Aula 23) pra adicionar, e delegação de clique no <ul> pra alternar concluida.

Solução

function carregarTarefas() {
  const dados = localStorage.getItem('tarefas-completas');
  return dados ? JSON.parse(dados) : [];
}
function salvarTarefas() {
  localStorage.setItem('tarefas-completas', JSON.stringify(tarefas));
}

let tarefas = carregarTarefas();

function renderizar() {
  listaTarefas.innerHTML = '';
  tarefas.forEach((tarefa, indice) => {
    const li = document.createElement('li');
    li.textContent = tarefa.texto;
    li.dataset.indice = indice;
    if (tarefa.concluida) li.classList.add('concluida');
    listaTarefas.appendChild(li);
  });
}

form.addEventListener('submit', (event) => {
  event.preventDefault();
  if (inputTarefa.value) {
    tarefas.push({ texto: inputTarefa.value, concluida: false });
    salvarTarefas();
    inputTarefa.value = '';
    renderizar();
  }
});

listaTarefas.addEventListener('click', (event) => {
  if (event.target.tagName === 'LI') {
    const indice = Number(event.target.dataset.indice);
    tarefas[indice].concluida = !tarefas[indice].concluida;
    salvarTarefas();
    renderizar();
  }
});

renderizar();

Checklist da Migração:

✔ tarefas como objetos {texto, concluida}

✔ carregarTarefas() + salvarTarefas()

✔ submit com preventDefault

✔ delegação alterna concluida e SALVA de novo
